Section freshening_demo.

  Notation "'x'" := 0.
  Notation "'y'" := 1.
  Notation "'z'" := 2.
  Notation "'v'" := 3.

  Example idfn := (λ v, `v).
  Example input := (λ x, λ x, `z · `x).
  Example output := subst y idfn input.
  Example expected_output := (λ x, λ y, `z · `y).

  Compute subst y idfn input.

  Goal output = expected_output.
    reflexivity.
  Qed.

End freshening_demo.
