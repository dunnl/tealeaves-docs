Class TraversableMonad2

  `{∀ B, Return (T B)}
  `{∀ B, Join (T B)} :=

    xxx_functor :> Functor2 T;

    xxx_dist2_ret:
    ∀ (B V: Type) `{Applicative G},
      dist2 ∘ ret (T := T (G B)) (A := G V) =
        map (F := G) (ret (T := T B) (A := V));
    xxx_dist2_join:
    ∀ (B V: Type) `{Applicative G},
      dist2 ∘ join (T := T (G B)) (A := (G V)) =

          map2 (F := T) id (dist2);
  }.
